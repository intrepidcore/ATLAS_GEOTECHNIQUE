2.1 Modifier scripts/generate_variogram_plot.py pour produire une vraie courbe γ(h) :

```python
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import io

def spherical_model(h, nugget, sill, range_m):
    """Modèle sphérique — CHASSAGNEUX eq. référence"""
    c = sill - nugget
    h = np.asarray(h, dtype=float)
    gamma = np.where(
        h <= range_m,
        nugget + c * (1.5*(h/range_m) - 0.5*(h/range_m)**3),
        nugget + c
    )
    return gamma

# Après avoir récupéré nugget, sill, range_m, model_type depuis ai_variograms :
h_max = range_m * 1.6
h_vals = np.linspace(0, h_max, 200)
gamma_vals = spherical_model(h_vals, nugget, sill, range_m)

fig, ax = plt.subplots(figsize=(9, 4))
fig.patch.set_facecolor('#0F172A')
ax.set_facecolor('#0F172A')

# Courbe théorique
ax.plot(h_vals/1000, gamma_vals, color='#38BDF8', linewidth=2,
        label=f'Modèle {model_type}')

# Lignes d'annotation
ax.axhline(y=sill, color='#F59E0B', linestyle='--', alpha=0.7,
           label=f'Sill = {sill:.3f}')
ax.axhline(y=nugget, color='#94A3B8', linestyle=':', alpha=0.7,
           label=f'Nugget = {nugget:.3f}')
ax.axvline(x=range_m/1000, color='#34D399', linestyle='--', alpha=0.7,
           label=f'Portée = {range_m/1000:.1f} km')

# Style dark
ax.spines['bottom'].set_color('#334155')
ax.spines['left'].set_color('#334155')
ax.spines['top'].set_visible(False)
ax.spines['right'].set_visible(False)
ax.tick_params(colors='#94A3B8')

ax.yaxis.label.set_color('#94A3B8')
ax.set_xlabel('Distance (km)', fontsize=10)

ax.set_title(f'Variogramme — {parameter_id} {horizon}', color='#F8FAFC', fontsize=11)

# Export SVG
```

ÉTAPE 2 — VRAIE COURBE VARIOGRAMME DANS LE DRAWER
═══════════════════════════════════════════════════════════

Modifier scripts/generate_variogram_plot.py :

```python
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import io, json, sys, os
import psycopg2

DB_URL = os.environ.get('DATABASE_URL', 'postgresql://atlas:atlas@localhost:5432/atlas_clean')

def spherical_model(h, nugget, sill, range_m):
    c = sill - nugget
    h = np.asarray(h, dtype=float)
    return np.where(
        h <= range_m,
        nugget + c * (1.5*(h/range_m) - 0.5*(h/range_m)**3),
        sill
    )

def generate_svg(parameter_id: str, horizon: str) -> dict:
    conn = psycopg2.connect(DB_URL)
    cur = conn.cursor()
    cur.execute("""
        SELECT v.nugget, v.sill, v.range_m, v.model_type, v.loo_rmse,
               v.fit_quality, r.method, v.created_at::text
        FROM atlas.ai_variograms v
        JOIN atlas.ai_interpolation_runs r ON r.id = v.run_id
        WHERE v.parameter_id = %s
          AND LOWER(COALESCE(v.fit_quality->>'horizon_label','')) = LOWER(%s)
        ORDER BY v.created_at DESC
        LIMIT 1
    """, (parameter_id, horizon))
    row = cur.fetchone()
    conn.close()

    if not row:
        return {"error": f"No variogram for {parameter_id}/{horizon}", "rows": 0}

    nugget, sill, range_m, model_type, loo_rmse, fit_quality, method, created_at = row

    if not nugget or not sill or not range_m:
        return {"error": "Incomplete variogram parameters", "rows": 0}

    # Générer la courbe
    h_max = float(range_m) * 1.6
    h_vals = np.linspace(0, h_max, 300)
    gamma_vals = spherical_model(h_vals, float(nugget), float(sill), float(range_m))

    fig, ax = plt.subplots(figsize=(9, 4.5))
    fig.patch.set_facecolor('#0F172A')
    ax.set_facecolor('#1E293B')

    # Courbe principale
    ax.plot(h_vals/1000, gamma_vals, color='#38BDF8', linewidth=2.5,
            label=f'Modèle {model_type or "sphérique"}')

    # Zone sous la courbe
    ax.fill_between(h_vals/1000, gamma_vals, alpha=0.08, color='#38BDF8')

    # Annotations
    ax.axhline(y=float(sill), color='#F59E0B', linestyle='--',
               linewidth=1.5, alpha=0.8, label=f'Palier (Sill) = {sill:.4f}')
    ax.axhline(y=float(nugget), color='#94A3B8', linestyle=':',
               linewidth=1.5, alpha=0.7, label=f'Effet pépite (Nugget) = {nugget:.4f}')
    ax.axvline(x=float(range_m)/1000, color='#34D399', linestyle='--',
               linewidth=1.5, alpha=0.8, label=f'Portée = {range_m/1000:.1f} km')

    # Style axes
    for spine in ax.spines.values():
        spine.set_color('#334155')
    ax.tick_params(colors='#94A3B8', labelsize=9)
    ax.set_xlabel('Distance (km)', color='#94A3B8', fontsize=10)
    ax.set_ylabel('γ(h)', color='#94A3B8', fontsize=11)
    ax.set_title(f'Variogramme théorique — {parameter_id} · {horizon}',
                 color='#F8FAFC', fontsize=12, pad=12)

    # LOO RMSE
    if loo_rmse:
        ax.text(0.99, 0.97, f'LOO RMSE = {float(loo_rmse):.4f}',
                transform=ax.transAxes, ha='right', va='top', fontsize=9,
                color='#94A3B8',
                bbox=dict(boxstyle='round,pad=0.4', facecolor='#0F172A',
                          edgecolor='#334155', alpha=0.9))

    # Légende
    ax.legend(facecolor='#1E293B', edgecolor='#334155',
              labelcolor='#94A3B8', fontsize=9, loc='lower right')

    ax.grid(True, color='#1E293B', linewidth=0.5, alpha=0.5)

    # Export SVG
    buf = io.BytesIO()
    fig.savefig(buf, format='svg', bbox_inches='tight',
                facecolor='#0F172A')
    svg_content = buf.getvalue().decode('utf-8')
    plt.close(fig)

    return {
        "svg": svg_content,
        "rows": 1,
        "nugget": float(nugget),
        "sill": float(sill),
        "range_m": float(range_m),
        "loo_rmse": float(loo_rmse) if loo_rmse else None,
        "model_type": model_type,
        "created_at": created_at
    }

if __name__ == '__main__':
    param = sys.argv[1] if len(sys.argv) > 1 else 'eg_avg'
    horizon = sys.argv[2] if len(sys.argv) > 2 else 'H1'
    result = generate_svg(param, horizon)
    print(json.dumps(result))
```

```python
#!/usr/bin/env python3
"""Extrait élévation, pente, TWI depuis le DSM pour les 29407 mailles."""
import psycopg2, numpy as np
from tqdm import tqdm

DB_URL = 'postgresql://atlas:atlas@127.0.0.1:5432/atlas_clean'
DSM_TABLE = 'atlas.dsm_togo'  # à adapter selon résultat raster_columns

conn = psycopg2.connect(DB_URL)
cur = conn.cursor()

# Récupérer toutes les mailles
cur.execute("SELECT code, ST_AsText(ST_PointOnSurface(geom)) FROM atlas.mailles ORDER BY code")
mailles = cur.fetchall()

print(f"Traitement de {len(mailles)} mailles...")

batch = []
for code, centroid_wkt in tqdm(mailles):
    # Élévation au centroïde
    cur.execute(f"""
        SELECT ST_Value(rast, ST_SetSRID(ST_GeomFromText(%s), 4326))
        FROM {DSM_TABLE}
        WHERE ST_Intersects(rast, ST_SetSRID(ST_GeomFromText(%s), 4326))
        LIMIT 1
    """, (centroid_wkt, centroid_wkt))
    row = cur.fetchone()
    elevation = float(row[0]) if row and row[0] is not None else None
    batch.append((elevation, code))

    if len(batch) >= 500:
        cur.executemany("""
            INSERT INTO atlas.ai_maille_features_fast (maille_code, elevation_m, feature_version)
            VALUES (%s, %s, 2)
            ON CONFLICT (maille_code) DO UPDATE
            SET elevation_m = EXCLUDED.elevation_m,
                feature_version = 2
        """, [(b[0], b[1]) for b in batch])
        conn.commit()
        batch = []

if batch:
    cur.executemany("""
        INSERT INTO atlas.ai_maille_features_fast (maille_code, elevation_m, feature_version)
        VALUES (%s, %s, 2)
        ON CONFLICT (maille_code) DO UPDATE
        SET elevation_m = EXCLUDED.elevation_m, feature_version = 2
    """, [(b[0], b[1]) for b in batch])
    conn.commit()

conn.close()
print("Terminé.")
```
