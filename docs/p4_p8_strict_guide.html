Tests psql obligatoires
-- Gate 1 : aucun IP_derived négatif

WHERE parameter_id LIKE 'ip_derived_%' AND value < 0;
-- Attendu : 0

-- Gate 2 : cohérence IP_derived ≈ IP_direct sur zones mesurées
WITH measured AS (
  SELECT s.maille_code, AVG(ea.ip_generated) as ip_measured
  FROM sondages s JOIN echantillons e ON e.sondage_id=s.id
  JOIN essais_atterberg ea ON ea.echantillon_id=e.id
  WHERE ea.ip_generated IS NOT NULL AND e.depth_m = 1.5
  GROUP BY s.maille_code
),
derived AS (
  SELECT maille_code, value as ip_derived
  FROM atlas.v_latest_ai_interpolation
  WHERE parameter_id = 'ip_derived_h2'
)
SELECT AVG(ABS(m.ip_measured - d.ip_derived)) as mae_vs_measured,
       MAX(ABS(m.ip_measured - d.ip_derived)) as max_error
FROM measured m JOIN derived d ON d.maille_code = m.maille_code;
-- Attendu : MAE < 5 % (acceptable pour kriging 87 pts)

R6-04La comparaison RGA_ked vs RGA_ancien (delta < 15 %) doit être vérifiée UNIQUEMENT sur les mailles avec sondages réels. Sur les mailles interpolées, l'ancien RGA était déjà rule-based — la comparaison ne valide pas la qualité du KED.
R6-05Stocker source_type='derived_rga_from_ked' distinct de l'ancien 'derived_rga_from_rules'. Les deux coexistent dans maille_geotech_infer pendant la période de validation.
